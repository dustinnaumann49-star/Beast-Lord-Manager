function runBattle() { function getTeam(prefix) { let team = []; for(let i = 1; i <= 3; i++) { let alpha = document.getElementById(`${prefix}-a-${i}`).value; let beastObj = beasts.find(o => o.name === document.getElementById(`${prefix}-b-${i}`).value); let count = parseInt(document.getElementById(`${prefix}-c-${i}`).value) || 0; team.push({ alpha: alpha, beast: beastObj.name, atk: beastObj.atk * count, def: beastObj.def * count, hp: beastObj.hp * count, type: beastObj.type, row: i }); } return team; } let teamA = getTeam("teamA"); let teamB = getTeam("teamB"); let log = "--- Kampf gestartet ---\n"; for (let r = 1; r <= 10; r++) { let logRunde = `Runde ${r}: `; // Schaden berechnen für Team A let atkA = teamA.reduce((s, t) => s + t.atk, 0); let bonusA = 1.0; if (teamA.some(t => t.alpha === "Hyaenodon" && t.type === "Fleischfresser") && Math.random() < 0.7) { bonusA = 1.15; logRunde += "[Hyaenodon: Nachahmung] "; } let dmgToB = Math.max(0, (atkA * bonusA) - (teamB.reduce((s,t) => s + t.def, 0) / 5)); // Schaden berechnen für Team B let atkB = teamB.reduce((s, t) => s + t.atk, 0); let bonusB = 1.0; if (teamB.some(t => t.alpha === "Hyaenodon" && t.type === "Fleischfresser") && Math.random() < 0.7) { bonusB = 1.15; } let dmgToA = Math.max(0, (atkB * bonusB) - (teamA.reduce((s,t) => s + t.def, 0) / 5)); // HP abziehen teamB.forEach(t => t.hp -= (dmgToB / 3)); teamA.forEach(t => t.hp -= (dmgToA / 3)); log += `${logRunde} Dein Schaden: ${Math.floor(dmgToB).toLocaleString()} | Gegner Schaden: ${Math.floor(dmgToA).toLocaleString()}\n`; if (teamA.reduce((s, t) => s + t.hp, 0) <= 0 || teamB.reduce((s, t) => s + t.hp, 0) <= 0) break; } document.getElementById('result').innerText = log + "\n--- Kampf beendet ---"; }
