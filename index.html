function runBattle() { function getStats(prefix) { let atk = 0, def = 0, hp = 0; for(let i=1; i<=3; i++) { let b = beasts.find(o => o.name === document.getElementById(`${prefix}-b-${i}`).value); let count = parseInt(document.getElementById(`${prefix}-c-${i}`).value) || 0; atk += b.atk * count; def += b.def * count; hp += b.hp * count; } return {atk, def, hp}; } let A = getStats("teamA"), B = getStats("teamB"); let log = "--- Kampf beginnt ---\n"; for(let r=1; r<=10; r++) { // Schaden: Angriff minus 20% der Verteidigung des Gegners let dmgToB = Math.max(0, A.atk - (B.def * 0.2)); let dmgToA = Math.max(0, B.atk - (A.def * 0.2)); A.hp -= dmgToA; B.hp -= dmgToB; log += `Runde ${r}: Dein Schaden: ${Math.floor(dmgToB).toLocaleString()} | Gegner Schaden: ${Math.floor(dmgToA).toLocaleString()}\n`; if(A.hp <= 0 || B.hp <= 0) break; } document.getElementById('result').innerText = log + "\n--- Kampf beendet ---"; }
